def scrape_pwc_trending(max_results: int = 10) -> List[Dict]:
    """用 Selenium 爬取 paperswithcode.com trending 页面."""
    driver = None
    try:
        driver = _get_driver()
        driver.get("https://paperswithcode.com/")
        time.sleep(3)  # JS 渲染
        # 用 Selenium 定位元素代替 regex
        cards = driver.find_elements(By.CSS_SELECTOR, ".paper-card, [class*='paper-card']")
        papers = []
        for card in cards[:max_results]:
            try:
                title_el = card.find_element(By.CSS_SELECTOR, "h1 a")
                title = title_el.text.strip()
                # 找 arXiv 链接
                links = card.find_elements(By.TAG_NAME, "a")
                arxiv_id = None
                for link in links:
                    href = link.get_attribute("href") or ""
                    if "arxiv.org" in href:
                        arxiv_id = _extract_arxiv_id(href)
                papers.append({"title": title, "arxiv_id": arxiv_id, ...})
            except Exception:
                continue
        return papers
    finally:
        if driver:
            driver.quit()

def check_selenium_available() -> bool:
    """快速检查 Selenium + Chrome/Edge 是否可用."""
    try:
        driver = _get_driver()
        driver.quit()
        return True
    except Exception:
        return False
